cat <<'EOF' | kubectl apply -f -
apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: scratch-data
spec:
  accessModes: ["ReadWriteOnce"]
  storageClassName: fast-ssd
  resources:
    requests:
      storage: 100Mi
---
apiVersion: v1
kind: Pod
metadata:
  name: scratch-writer
spec:
  containers:
    - name: writer
      image: busybox:1.37
      command: ["sh", "-c", "date > /data/started && sleep 3600"]
      volumeMounts:
        - name: data
          mountPath: /data
  volumes:
    - name: data
      persistentVolumeClaim:
        claimName: scratch-data
EOF
---
persistentvolumeclaim/scratch-data created
pod/scratch-writer created
# exit 0
